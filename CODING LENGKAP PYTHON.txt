# ===== FUNGSI UTILITY =====
# Fungsi untuk mengaktifkan relay (Active HIGH)
def set_relay(relay_num, state):
    if relay_num < 1 or relay_num > 4:
        return
    
    # Simulasi kontrol relay (dalam implementasi nyata, ini akan mengontrol GPIO)
    relay_status[relay_num - 1] = state
    
    print(f"Relay {relay_num} -> {'ON (HIGH)' if state else 'OFF (LOW)'}")

# Fungsi untuk mengaktifkan relay dengan timer
def set_relay_with_timer(relay_num, duration):
    set_relay(relay_num, True)
    relay_start_time[relay_num - 1] = time.time()
    relay_timer_active[relay_num - 1] = True
    
    print(f"Timer Relay {relay_num} aktif: {duration} detik")
    
    # Start a timer thread to turn off the relay after duration seconds
    def turn_off_relay():
        time.sleep(duration)
        deactivate_relay(relay_num)
    
    thread = threading.Thread(target=turn_off_relay)
    thread.daemon = True
    thread.start()

# Fungsi untuk menonaktifkan relay dan mengecek apakah semua sudah selesai
def deactivate_relay(relay_num):
    global drink_in_progress, active_relay, drink_finish_time, waiting_to_return_servo
    
    set_relay(relay_num, False)
    relay_timer_active[relay_num - 1] = False
    
    # Cek apakah semua relay sudah mati
    all_off = all(not status for status in relay_status)
    
    # Jika semua relay mati dan ada minuman dalam proses
    if all_off and drink_in_progress:
        drink_in_progress = False
        active_relay = 0
        
        # Set waktu untuk menunggu 2 detik sebelum servo kembali
        drink_finish_time = time.time()
        waiting_to_return_servo = True
        
        print("Minuman selesai, servo akan kembali ke 0° dalam 2 detik")

# Fungsi untuk menggerakkan servo
def move_servo(position):
    global servo_position
    # Simulasi kontrol servo (dalam implementasi nyata, ini akan mengontrol servo melalui GPIO)
    servo_position = position
    
    print(f"Servo bergerak ke posisi: {position} derajat")

# Fungsi untuk mematikan semua relay (EMERGENCY)
def emergency_stop():
    global drink_in_progress, active_relay, servo_ready
    
    print("EMERGENCY STOP!")
    for i in range(1, 5):
        set_relay(i, False)
        relay_timer_active[i - 1] = False
    
    # Reset status proses
    drink_in_progress = False
    active_relay = 0
    
    # Kembalikan servo ke posisi awal
    if servo_position != 0:
        move_servo(0)
        servo_ready = False

# Fungsi untuk menangani tombol fisik
def handle_physical_button():
    global button_pressed, button_press_time, servo_ready, servo_ready_time
    
    # Simulasi tombol fisik (dalam implementasi nyata, ini akan membaca input GPIO)
    # Untuk simulasi, kita akan menggunakan fungsi ini sebagai placeholder
    
    # Jika tombol ditekan (LOW karena pull-up)
    if not button_pressed and time.time() - button_press_time > debounce_delay:
        button_pressed = True
        button_press_time = time.time()
        
        # Jika servo di posisi awal dan tidak ada proses minuman
        if servo_position == 0 and not drink_in_progress:
            print("Tombol fisik ditekan - memutar servo ke 90°")
            move_servo(90)
            servo_ready = True
            servo_ready_time = time.time()
    
    # Jika tombol dilepas
    if button_pressed:
        button_pressed = False

# Background thread to handle physical button and timers
def background_tasks():
    while True:
        # Handle physical button
        handle_physical_button()
        
        # Cek timeout servo
        if servo_ready and not drink_in_progress:
            if time.time() - servo_ready_time >= servo_timeout:
                # Timeout, kembalikan servo ke posisi awal
                print("Servo timeout - kembali ke posisi 0°")
                move_servo(0)
                servo_ready = False
        
        # Cek apakah sudah waktunya mengembalikan servo setelah minuman selesai
        if waiting_to_return_servo and servo_position == 90:
            if time.time() - drink_finish_time >= 2:
                # Sudah 2 detik setelah minuman selesai
                print("2 detik telah berlalu - mengembalikan servo ke posisi 0°")
                move_servo(0)
                servo_ready = False
                waiting_to_return_servo = False
        
        time.sleep(0.01)  # Small delay for stability

# Handler untuk kontrol relay
@app.route('/relay')
def handle_relay():
    global drink_in_progress, active_relay, servo_ready, servo_position
    
    if 'num' in request.args and 'state' in request.args:
        relay_num = int(request.args['num'])
        state = int(request.args['state'])
        
        # Special commands
        if relay_num == 0:
            # Servo control
            if state == 1:
                # Rotate servo to 90
                move_servo(90)
                servo_ready = True
                servo_ready_time = time.time()
            else:
                # Return servo to 0
                move_servo(0)
                servo_ready = False
            return "OK"
        elif relay_num == 99:
            # Emergency stop
            emergency_stop()
            return "OK"
        
        if 1 <= relay_num <= 4:
            if state == 1:
                # Cek ketersediaan menu
                if not drink_available[relay_num - 1]:
                    return "Menu not available", 400
                
                # Cek apakah ada parameter duration (untuk test pump)
                is_test_mode = 'duration' in request.args
                duration = int(request.args['duration']) if is_test_mode else relay_timers[relay_num - 1]
                
                # Jika bukan test mode, cek kondisi servo dan proses lain
                if not is_test_mode:
                    # Check if servo is ready
                    if not servo_ready:
                        return "Servo not ready", 400
                    
                    # Check if another drink is in progress
                    if drink_in_progress and active_relay != relay_num:
                        return "Another drink in progress", 400
                
                # Aktifkan relay dengan timer
                set_relay_with_timer(relay_num, duration)
                
                # Set drink in progress hanya jika bukan test mode
                if not is_test_mode:
                    drink_in_progress = True
                    active_relay = relay_num
                
                print(f"{'TEST MODE - ' if is_test_mode else ''}Relay {relay_num} ON untuk {duration} detik")
            else:
                # Matikan relay dan cek apakah semua sudah selesai
                deactivate_relay(relay_num)
            
            return "OK"
        else:
            return "Invalid relay number", 400
    else:
        return "Missing parameters", 400

@app.route('/updateSettings')
def handle_update_settings():
    if 'num' in request.args and 'sec' in request.args and 'price' in request.args and \
       'name' in request.args and 'icon' in request.args and 'available' in request.args:
        relay_num = int(request.args['num'])
        seconds = int(request.args['sec'])
        price = int(request.args['price'])
        name = request.args['name']
        icon = request.args['icon']
        available = request.args['available'] == '1'
        
        if 1 <= relay_num <= 4 and seconds > 0 and price >= 0 and name:
            relay_timers[relay_num - 1] = seconds
            drink_prices[relay_num - 1] = price
            drink_names[relay_num - 1] = name
            drink_icons[relay_num - 1] = icon
            drink_available[relay_num - 1] = available
            
            print(f"Update Menu {relay_num}: Nama={name}, Icon={icon}, Timer={seconds}s, Harga=Rp {price}, Status={'Tersedia' if available else 'Tidak Tersedia'}")
            
            return "OK"
        else:
            return "Invalid parameters", 400
    else:
        return "Missing parameters", 400
