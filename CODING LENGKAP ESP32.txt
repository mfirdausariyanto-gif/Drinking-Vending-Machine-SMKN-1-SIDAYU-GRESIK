// ===== FUNGSI UTILITY =====
// Fungsi untuk mengaktifkan relay (Active HIGH)
void setRelay(int relayNum, bool state) {
  int pin = 0;
  switch(relayNum) {
    case 1: pin = RELAY_PIN_1; break;
    case 2: pin = RELAY_PIN_2; break;
    case 3: pin = RELAY_PIN_3; break;
    case 4: pin = RELAY_PIN_4; break;
    default: return;
  }
  
  // Active HIGH: ON = HIGH, OFF = LOW
  digitalWrite(pin, state ? HIGH : LOW);
  relayStatus[relayNum - 1] = state;
  
  Serial.print("Relay ");
  Serial.print(relayNum);
  Serial.print(" -> ");
  Serial.println(state ? "ON (HIGH)" : "OFF (LOW)");
}

// Fungsi untuk mengaktifkan relay dengan timer
void setRelayWithTimer(int relayNum, int duration) {
  setRelay(relayNum, true);
  relayStartTime[relayNum - 1] = millis();
  relayTimerActive[relayNum - 1] = true;
  
  Serial.print("Timer Relay ");
  Serial.print(relayNum);
  Serial.print(" aktif: ");
  Serial.print(duration);
  Serial.println(" detik");
}

// Fungsi untuk menonaktifkan relay dan mengecek apakah semua sudah selesai
void deactivateRelay(int relayNum) {
  setRelay(relayNum, false);
  relayTimerActive[relayNum - 1] = false;
  
  // Cek apakah semua relay sudah mati
  bool allOff = true;
  for (int i = 0; i < 4; i++) {
    if (relayStatus[i]) {
      allOff = false;
      break;
    }
  }
  
  // Jika semua relay mati dan ada minuman dalam proses
  if (allOff && drinkInProgress) {
    drinkInProgress = false;
    activeRelay = 0;
    
    // Set waktu untuk menunggu 2 detik sebelum servo kembali
    drinkFinishTime = millis();
    waitingToReturnServo = true;
    
    Serial.println("Minuman selesai, servo akan kembali ke 0° dalam 2 detik");
  }
}

// Fungsi untuk menggerakkan servo
void moveServo(int position) {
  cupServo.write(position);
  servoPosition = position;
  
  Serial.print("Servo bergerak ke posisi: ");
  Serial.print(position);
  Serial.println(" derajat");
}

// Fungsi untuk mematikan semua relay (EMERGENCY)
void emergencyStop() {
  Serial.println("EMERGENCY STOP!");
  for(int i = 1; i <= 4; i++) {
    setRelay(i, false);
    relayTimerActive[i - 1] = false;
  }
  
  // Reset status proses
  drinkInProgress = false;
  activeRelay = 0;
  
  // Kembalikan servo ke posisi awal
  if (servoPosition != 0) {
    moveServo(0);
    servoReady = false;
  }
}

// Fungsi untuk menangani tombol fisik
void handlePhysicalButton() {
  int reading = digitalRead(BUTTON_PIN);
  
  // Jika tombol ditekan (LOW karena pull-up)
  if (reading == LOW && !buttonPressed && millis() - buttonPressTime > debounceDelay) {
    buttonPressed = true;
    buttonPressTime = millis();
    
    // Jika servo di posisi awal dan tidak ada proses minuman
    if (servoPosition == 0 && !drinkInProgress) {
      Serial.println("Tombol fisik ditekan - memutar servo ke 90°");
      moveServo(90);
      servoReady = true;
      servoReadyTime = millis();
    }
  }
  
  // Jika tombol dilepas
  if (reading == HIGH && buttonPressed) {
    buttonPressed = false;
  }
}

// Handler untuk kontrol relay
void handleRelay() {
  if (server.hasArg("num") && server.hasArg("state")) {
    int relayNum = server.arg("num").toInt();
    int state = server.arg("state").toInt();
    
    // Special commands
    if (relayNum == 0) {
      // Servo control
      if (state == 1) {
        // Rotate servo to 90
        moveServo(90);
        servoReady = true;
        servoReadyTime = millis();
      } else {
        // Return servo to 0
        moveServo(0);
        servoReady = false;
      }
      server.send(200, "text/plain", "OK");
      return;
    } else if (relayNum == 99) {
      // Emergency stop
      emergencyStop();
      server.send(200, "text/plain", "OK");
      return;
    }
    
    if (relayNum >= 1 && relayNum <= 4) {
      if (state == 1) {
        // Cek ketersediaan menu
        if (!drinkAvailable[relayNum - 1]) {
          server.send(400, "text/plain", "Menu not available");
          return;
        }
        
        // Cek apakah ada parameter duration (untuk test pump)
        bool isTestMode = server.hasArg("duration");
        int duration = isTestMode ? server.arg("duration").toInt() : relayTimers[relayNum - 1];
        
        // Jika bukan test mode, cek kondisi servo dan proses lain
        if (!isTestMode) {
          // Check if servo is ready
          if (!servoReady) {
            server.send(400, "text/plain", "Servo not ready");
            return;
          }
          
          // Check if another drink is in progress
          if (drinkInProgress && activeRelay != relayNum) {
            server.send(400, "text/plain", "Another drink in progress");
            return;
          }
        }
        
        // Aktifkan relay dengan timer
        setRelayWithTimer(relayNum, duration);
        
        // Set drink in progress hanya jika bukan test mode
        if (!isTestMode) {
          drinkInProgress = true;
          activeRelay = relayNum;
        }
        
        Serial.print(isTestMode ? "TEST MODE - " : "");
        Serial.print("Relay ");
        Serial.print(relayNum);
        Serial.print(" ON untuk ");
        Serial.print(duration);
        Serial.println(" detik");
      } else {
        // Matikan relay dan cek apakah semua sudah selesai
        deactivateRelay(relayNum);
      }
      
      server.send(200, "text/plain", "OK");
    } else {
      server.send(400, "text/plain", "Invalid relay number");
    }
  } else {
    server.send(400, "text/plain", "Missing parameters");
  }
}

// Handler untuk update settings (timer, harga, nama, icon, dan ketersediaan)
void handleUpdateSettings() {
  if (server.hasArg("num") && server.hasArg("sec") && server.hasArg("price") && 
      server.hasArg("name") && server.hasArg("icon") && server.hasArg("available")) {
    int relayNum = server.arg("num").toInt();
    int seconds = server.arg("sec").toInt();
    int price = server.arg("price").toInt();
    String name = server.arg("name");
    String icon = server.arg("icon");
    bool available = server.arg("available").toInt() == 1;
    
    if (relayNum >= 1 && relayNum <= 4 && seconds > 0 && price >= 0 && name.length() > 0) {
      relayTimers[relayNum - 1] = seconds;
      drinkPrices[relayNum - 1] = price;
      drinkNames[relayNum - 1] = name;
      drinkIcons[relayNum - 1] = icon;
      drinkAvailable[relayNum - 1] = available;
      
      Serial.print("Update Menu ");
      Serial.print(relayNum);
      Serial.print(": Nama=");
      Serial.print(name);
      Serial.print(", Icon=");
      Serial.print(icon);
      Serial.print(", Timer=");
      Serial.print(seconds);
      Serial.print("s, Harga=Rp ");
      Serial.print(price);
      Serial.print(", Status=");
      Serial.println(available ? "Tersedia" : "Tidak Tersedia");
      
      server.send(200, "text/plain", "OK");
    } else {
      server.send(400, "text/plain", "Invalid parameters");
    }
  } else {
    server.send(400, "text/plain", "Missing parameters");
  }
}

// ===== SETUP =====
void setup() {
  // Inisialisasi Serial Monitor
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n\n=================================");
  Serial.println("Smart Dispenser ESP32 Starting...");
  Serial.println("=================================\n");
  
  // Setup GPIO pins sebagai OUTPUT
  pinMode(RELAY_PIN_1, OUTPUT);
  pinMode(RELAY_PIN_2, OUTPUT);
  pinMode(RELAY_PIN_3, OUTPUT);
  pinMode(RELAY_PIN_4, OUTPUT);
  
  // Setup pin tombol fisik dengan pull-up internal
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  
  // Matikan semua relay saat startup (Active HIGH: LOW = OFF)
  digitalWrite(RELAY_PIN_1, LOW);
  digitalWrite(RELAY_PIN_2, LOW);
  digitalWrite(RELAY_PIN_3, LOW);
  digitalWrite(RELAY_PIN_4, LOW);
  
  // Inisialisasi servo
  ESP32PWM::allocateTimer(0);
  ESP32PWM::allocateTimer(1);
  ESP32PWM::allocateTimer(2);
  ESP32PWM::allocateTimer(3);
  cupServo.setPeriodHertz(50);
  cupServo.attach(SERVO_PIN, 500, 2400);
  moveServo(0);
  
  Serial.println("Mode: ACTIVE HIGH");
  Serial.println("  HIGH = Relay ON");
  Serial.println("  LOW  = Relay OFF");
  Serial.println("  Tombol Fisik = Pull-up (HIGH saat idle, LOW saat ditekan)");
  
  Serial.println("Pin GPIO dikonfigurasi:");
  Serial.println("  Relay 1 (Nutrisari)  -> GPIO 19");
  Serial.println("  Relay 2 (Lemon Tea)  -> GPIO 18");
  Serial.println("  Relay 3 (Cappucino)  -> GPIO 17");
  Serial.println("  Relay 4 (Milo)       -> GPIO 16");
  Serial.println("  Servo Motor         -> GPIO 13");
  Serial.println("  Tombol Fisik        -> GPIO 12");
  Serial.println();
  
  // Koneksi ke WiFi
  Serial.print("Menghubungkan ke WiFi: ");
  Serial.println(ssid);
  
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 30) {
    delay(500);
    Serial.print(".");
    attempts++;
  }
  
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n\n✓ WiFi Terhubung!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());
    Serial.print("Signal Strength: ");
    Serial.print(WiFi.RSSI());
    Serial.println(" dBm");
  } else {
    Serial.println("\n\n✗ Gagal terhubung ke WiFi!");
    Serial.println("Periksa SSID dan Password Anda");
  }
  
  // Setup Web Server routes
  server.on("/", handleRoot);

// ===== LOOP UTAMA =====
void loop() {
  // Handle client requests
  server.handleClient();
  
  // Handle physical button
  handlePhysicalButton();
  
  // Cek timer relay untuk auto-off
  unsigned long currentMillis = millis();
  
  for(int i = 0; i < 4; i++) {
    if (relayTimerActive[i]) {
      // Cek apakah timer sudah habis
      unsigned long elapsed = currentMillis - relayStartTime[i];
      
      if (elapsed >= (relayTimers[i] * 1000)) {
        // Timer habis, matikan relay
        Serial.print("Timer Relay ");
        Serial.print(i + 1);
        Serial.println(" selesai - AUTO OFF");
        deactivateRelay(i + 1);
      }
    }
  }
  
  // Cek timeout servo
  if (servoReady && !drinkInProgress) {
    if (currentMillis - servoReadyTime >= servoTimeout) {
      // Timeout, kembalikan servo ke posisi awal
      Serial.println("Servo timeout - kembali ke posisi 0°");
      moveServo(0);
      servoReady = false;
    }
  }
  
  // Cek apakah sudah waktunya mengembalikan servo setelah minuman selesai
  if (waitingToReturnServo && servoPosition == 90) {
    if (currentMillis - drinkFinishTime >= 2000) {
      // Sudah 2 detik setelah minuman selesai
      Serial.println("2 detik telah berlalu - mengembalikan servo ke posisi 0°");
      moveServo(0);
      servoReady = false;
      waitingToReturnServo = false;
    }
  }
 
  // Small delay untuk stabilitas
  delay(10);
}
