    function controlRelay(relayNum, state, duration) {
      if (!drinkAvailable[relayNum]) {
        addLog('⚠️ Gagal: ' + drinkNames[relayNum] + ' tidak tersedia!', 'warning');
        return;
      }
      if (state && drinkInProgress && activeRelay !== relayNum) {
        addLog('⚠️ Gagal: ' + drinkNames[relayNum] + '. Minuman lain sedang diproses!', 'warning');
        return;
      }
      if (state && !servoReadyForDrink) {
        addLog('⚠️ Gagal: ' + drinkNames[relayNum] + '. Tekan tombol fisik terlebih dahulu!', 'warning');
        return;
      }
      relayStates[relayNum] = state;
      updateRelayStatus(relayNum, state);
      if (state) {
        const timerDuration = duration || relayTimers[relayNum];
        addLog('✓ ' + drinkNames[relayNum] + ' dipilih - Proses dimulai (' + timerDuration + 's)', 'success');
        drinkInProgress = true;
        activeRelay = relayNum;
        updateServoStatus('processing');
        updateDrinkCards();
        document.getElementById('processIndicator').classList.remove('active');
        if (activeTimers[relayNum]) clearTimeout(activeTimers[relayNum]);
        activeTimers[relayNum] = setTimeout(function() {
          controlRelay(relayNum, false);
          addLog('✓ ' + drinkNames[relayNum] + ' selesai', 'success');
        }, timerDuration * 1000);
      } else {
        if (!activeTimers[relayNum]) addLog('✗ ' + drinkNames[relayNum] + ' dihentikan', 'error');
        if (activeTimers[relayNum]) {
          clearTimeout(activeTimers[relayNum]);
          delete activeTimers[relayNum];
        }
        let allOff = true;
        for (let i = 1; i <= 4; i++) {
          if (relayStates[i]) {
            allOff = false;
            break;
          }
        }
        if (allOff && drinkInProgress) {
          drinkInProgress = false;
          activeRelay = 0;
          waitingToReturnServo = true;
          addLog('Semua minuman selesai, servo akan kembali dalam 2 detik');
          setTimeout(function() {
            if (servoPosition === 90 && waitingToReturnServo) {
              returnServoToZero();
              waitingToReturnServo = false;
            }
          }, 2000);
        }
        updateDrinkCards();
      }
      sendToESP32(relayNum, state, duration);
    }

    function sendToESP32(relay, state, duration) {
      let url = `${API_URL}/relay?num=${relay}&state=${state ? '1' : '0'}`;
      if (duration) url += `&duration=${duration}`;
      fetch(url)
        .then(response => {
          if (response.ok) {
            addLog('✓ Perintah dikirim ke sistem', 'success');

            connectionErrorCount = 0;

    }

        relayTimers[i] = seconds;
        drinkPrices[i] = price;
        drinkNames[i] = name;

        updatePriceDisplay(i, price);

    function addLog(message, type = 'info') {

      logEntry.className = 'log-entry';

    function startPolling() {
      if (pollingInterval) return;
      pollingEnabled = true;
      pollingInterval = setInterval(function() {
        if (!pollingEnabled) return;
        fetch(`${API_URL}/status`)
          .then(response => response.json())
          .then(data => {
            isConnected = true;
            connectionErrorCount = 0;
            for (let i = 1; i <= 4; i++) {
              if (data['relay' + i] !== undefined) {
                const newState = data['relay' + i];
                if (relayStates[i] !== newState) {
                  relayStates[i] = newState;
                  updateRelayStatus(i, newState);
                }
              }
              if (data['available' + i] !== undefined) {
                drinkAvailable[i] = data['available' + i];
              }
            }
            if (data.servoPosition !== undefined && servoPosition !== data.servoPosition) {
              servoPosition = data.servoPosition;
              document.getElementById('servoPosition').textContent = servoPosition + '°';
              document.getElementById('rotateServoBtn').disabled = (servoPosition !== 0 || drinkInProgress);
              document.getElementById('resetServoBtn').disabled = (servoPosition !== 90);
              document.getElementById('manualButton').disabled = (servoPosition !== 0 || drinkInProgress);
              servoReadyForDrink = (servoPosition === 90);
              const indicator = document.getElementById('processIndicator');
              if (servoReadyForDrink) {
                indicator.textContent = 'Servo siap - Silakan pilih minuman';
                indicator.classList.add('active');
              } else {
                indicator.classList.remove('active');
              }
              updateServoStatus(servoReadyForDrink ? 'waiting' : 'ready');
              updateDrinkCards();
            }
            if (data.drinkInProgress !== undefined) drinkInProgress = data.drinkInProgress;
            if (data.activeRelay !== undefined) activeRelay = data.activeRelay;
          })
          .catch(error => {
            connectionErrorCount++;
            if (connectionErrorCount >= 3) {
              stopPolling();
              addLog('⚠️ Koneksi terputus - Polling dihentikan', 'warning');
            }
          });
      }, 1000);
    }

    function stopPolling() {
      pollingEnabled = false;
      if (pollingInterval) {
        clearInterval(pollingInterval);
        pollingInterval = null;
      }
    }
